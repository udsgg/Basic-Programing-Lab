```C
#include <stdio.h>
#define SIZE 3

int main() {
    int data[SIZE];
    int *ptr;
    int i;

    printf("Enter %d integer elements for the array:\n", SIZE);
    for (i = 0; i < SIZE; i++) {
        scanf("%d", &data[i]);
    }
    ptr = data;

    printf("\n---ARRAY ANDPOINTER REPORT---\n");
    printf("Index | Value(Direct) | Value(Pointer) | Address\n");

    for (i = 0; i < SIZE; i++) {
        printf("%2d | %4d | %4d | %p\n", 
               i, data[i], (*(ptr + i)), (ptr + i));
    }
    return 0;

}
```
