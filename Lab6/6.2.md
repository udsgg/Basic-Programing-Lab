```C
#include <stdio.h>

void swap_value(int *a, int *b);

int main() {
    int num1, num2;

    printf("Enter value for num 1: ");
    scanf("%d", &num1);
    printf("Enter value for num 2: ");
    scanf("%d", &num2);

    printf("\n---SWAP VALUE REPORT---\n");
    printf("BEFORE swap - num1 = %d, num2 = %d\n", num1, num2);
    swap_value(&num1, &num2);
    printf("AFTER swap - num1 = %d, num2 = %d\n", num1, num2);
}

void swap_value(int *a, int *b) {
    if (a != NULL && b != NULL) {
        int temp = *a;
        *a = *b;
        *b = temp;
    }
}
```
